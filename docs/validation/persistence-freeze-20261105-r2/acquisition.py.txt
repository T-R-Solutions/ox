from pathlib import Path
import json,tarfile,subprocess,hashlib
b=Path('docs/validation/persistence-freeze-20261105-r2/upstream');archives=Path('.local/build/persistence-experiment-preflight-20261105/reviewed-crates'); requests=set()
for p in archives.glob('iced_*.crate'):
 with tarfile.open(p) as t:
  v=json.load(t.extractfile(p.stem+'/.cargo_vcs_info.json'));rev=v['git']['sha1'];sub=v['path_in_vcs']
  requests.update((rev,n) for n in ['LICENSE','Cargo.toml',sub+'/Cargo.toml',sub+'/LICENSE'])
rows=[]
for rev,n in sorted(requests):
 out=b/rev/n;out.parent.mkdir(parents=True,exist_ok=True);url=f'https://raw.githubusercontent.com/iced-rs/iced/{rev}/{n}'
 p=subprocess.run(['curl','--silent','--show-error','--location','--max-time','30','--dump-header',str(out)+'.headers','--output',str(out),'--write-out','%{http_code}',url],capture_output=True,text=True)
 rows.append({'url':url,'path':str(out),'http_status':p.stdout,'exit_code':p.returncode,'stderr':p.stderr,'sha256':hashlib.sha256(out.read_bytes()).hexdigest() if out.exists() else None})
(b/'acquisition.json').write_text(json.dumps(rows,indent=2)+'\n');print([(r['url'],r['http_status'],r['exit_code']) for r in rows])
