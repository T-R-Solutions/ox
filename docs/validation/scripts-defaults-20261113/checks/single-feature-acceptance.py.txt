from pathlib import Path
import importlib.util
import sys

root = Path('/home/rodrigo/Projetos/ox')
spec = importlib.util.spec_from_file_location('ox_sdd_validator', root / 'sdd/tools/validate.py')
validator = importlib.util.module_from_spec(spec)
spec.loader.exec_module(validator)
feature = root / 'sdd/features/scripts-centralization'
document = validator.strict_json(feature / 'evidence.json')
rows = validator.strict_json(feature / 'traceability.json')
errors = validator.evidence_errors(document, root)
errors += validator.trace_errors(rows, root, feature, document)
errors += validator.input_errors(document, root, rows, feature, True)
expected = validator.acceptance_ids((feature / 'spec.md').read_text())
for row in document['requirements']:
    if row['id'] in expected and validator.state(row['status']) not in {'APPROVED', 'NOT APPLICABLE'}:
        errors.append('Unapproved criterion: ' + row['id'])
for error in errors:
    print('FAIL:', error)
if not errors:
    print('PASS: scripts-centralization criteria, evidence schema, traceability and fresh input bindings. No product acceptance.')
sys.exit(bool(errors))
